% ---------------------------------------------------------------- Captions & numbering
% Figures, tables (and equations) always carry the chapter number, chapter 0 included,
% so that PDF and HTML numbering agree (the filter numbers HTML the same way).
\renewcommand*{\thefigure}{\thechapter.\arabic{figure}}
\renewcommand*{\thetable}{\thechapter.\arabic{table}}
\renewcommand*{\captionformat}{\nobreakspace:\ }
\AtBeginDocument{\renewcommand{\tablename}{Tableau}\renewcommand{\figurename}{Figure}}
% Unnumbered chapters (preface, conclusion): the filter makes their sections unnumbered and
% calls \numerotationlocale{<tag>} before the chapter, so figures, tables and equations are
% numbered 1, 2… as in HTML (with distinct PDF anchors); \numerotationchapitre, called before
% every numbered chapter, restores the chapter-prefixed numbering saved at \begin{document}.
\makeatletter
\AtBeginDocument{%
  \let\mono@thefigure\thefigure \let\mono@thetable\thetable \let\mono@theequation\theequation
  \let\mono@theHfigure\theHfigure \let\mono@theHtable\theHtable
  \let\mono@theHequation\theHequation}
\newcommand*{\numerotationlocale}[1]{%
  \setcounter{figure}{0}\setcounter{table}{0}\setcounter{equation}{0}%
  \renewcommand*{\thefigure}{\arabic{figure}}\renewcommand*{\thetable}{\arabic{table}}%
  \renewcommand*{\theequation}{\arabic{equation}}%
  \renewcommand*{\theHfigure}{#1.\arabic{figure}}\renewcommand*{\theHtable}{#1.\arabic{table}}%
  \renewcommand*{\theHequation}{#1.\arabic{equation}}}
\newcommand*{\numerotationchapitre}{%
  \let\thefigure\mono@thefigure \let\thetable\mono@thetable \let\theequation\mono@theequation
  \let\theHfigure\mono@theHfigure \let\theHtable\mono@theHtable
  \let\theHequation\mono@theHequation}
\makeatother

% ---------------------------------------------------------------- Front matter
% Pandoc's template opens the front matter before the title and closes it with \mainmatter
% right after the table of contents. The preface (filter, class .avant-propos) reopens it
% with \reprendreliminaires, which carries on the roman page numbers of the TOC; a second
% \frontmatter would restart them at i (duplicate page numbers and PDF anchors).
% KOMA's \frontmatter and \mainmatter clear the page, then call \pagenumbering, which
% resets the counter: the hook keeps the number the next page would have had.
\newcounter{pagesuivante}
\AddToHook{cmd/pagenumbering/before}{\setcounter{pagesuivante}{\value{page}}}
\newcommand{\reprendreliminaires}{%
  \edef\pagereprise{\arabic{pagesuivante}}%
  \frontmatter
  \setcounter{page}{\pagereprise}}

% (filter, pass 6), so its length is free; the part's first chapter opens a new page.
